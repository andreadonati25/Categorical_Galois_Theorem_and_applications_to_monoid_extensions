\begin{frame}{Pullback functor}
    Let $\sigma:S\to R$ be a morphism in a category $\mathcal{A}$ with pullbacks. 
    
    The functor
    \[
    \begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&, row sep=1pt]
        \sigma^*:{\mathcal{A}}/R \arrow[r] \& {\mathcal{A}}/S \\
        a:A\to R \arrow[r] \& p_S:S\times_RA\to S
    \end{tikzcd}
    \]
    
    %\pause
    \vspace{4mm}
    
    has a left adjoint functor
    \[
    \begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&, row sep=1pt]
        \Sigma_\sigma:{\mathcal{A}}/S \arrow[r] \& {\mathcal{A}}/R \\
        a:A\to S \arrow[r] \& \sigma\circ a:A\to R.
    \end{tikzcd}
    \]
\end{frame}

% 2 %
\begin{frame}{Induced adjunction}
    Let $\begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&]
        \mathcal{A} \arrow[r, bend left, "\mathcal{S}", ""{name=S}] \& \mathcal{P} \arrow[l, bend left, "\mathcal{C}", ""{name=C}] \arrow[phantom, from=S, to=C, "\dashv" rotate=-90]
    \end{tikzcd}$ be an adjunction where $\mathcal{A}$ has pullbacks.
    
    For every object $S\in\mathcal{A}$, the functor
    \[
    \begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&, row sep=1pt]
        \mathcal{S}_S:{\mathcal{A}}/S \arrow[r] \& {\mathcal{P}}/\mathcal{S}(S) \\
        a:A\to S \arrow[r] \& \mathcal{S}(a):\mathcal{S}(A)\to \mathcal{S}(S)
    \end{tikzcd}
    \]

    %\pause
    \vspace{3mm}
    
    has a right adjoint functor
    \[
    \begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&, row sep=1pt]
        \mathcal{C}_S:{\mathcal{P}}/\mathcal{S}(S) \arrow[r] \& {\mathcal{A}}/S \\
        f:X\to \mathcal{S}(S) \arrow[r] \& \eta_S^*(\mathcal{C}(f)):S\times_{\mathcal{C}\mathcal{S}(S)}\mathcal{C}(X)\to S
    \end{tikzcd}
    \]
    where $\eta_S$ is the unit of $\mathcal{S}\dashv\mathcal{C}$.
\end{frame}

% 3 %
\begin{frame}{Relatively admissible adjunction}
    \begin{defi}
        A relatively admissible adjunction 
        $\begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&]
            (\mathcal{A},\overline{\mathcal{A}}) \arrow[r, bend left, "\mathcal{S}", ""{name=S}] \& (\mathcal{P},\overline{\mathcal{P}}) \arrow[l, bend left, "\mathcal{C}", ""{name=C}] \arrow[phantom, from=S, to=C, "\dashv" rotate=-90]
        \end{tikzcd}$ consists in an adjunction $\mathcal{S}\dashv\mathcal{C}$ and two classes of arrows $\overline{\mathcal{A}}$, $\overline{\mathcal{P}}$ such that:
        
        %\pause
        \vspace{1mm}
        
        \begin{enumerate}
            \item $\overline{\mathcal{A}}$ and $\overline{\mathcal{P}}$ contain all isomorphisms;
            \item $\overline{\mathcal{A}}$ and $\overline{\mathcal{P}}$ are closed under composition;
            \item $\overline{\mathcal{A}}$ and $\overline{\mathcal{P}}$ are pullback-stable;
            \item $\mathcal{S}(\overline{\mathcal{A}})\subset\overline{\mathcal{P}}$;
            \item $\mathcal{C}(\overline{\mathcal{P}})\subset\overline{\mathcal{A}}$;
            \item $\eta_A\in\overline{\mathcal{A}}$, for every $A$ in $\mathcal{A}$;
            \item $\varepsilon_X\in\overline{\mathcal{P}}$, for every $X$ in $\mathcal{P}$.
        \end{enumerate}
    \end{defi}
\end{frame}

% 4 %
\begin{frame}{Relative Galois descent morphism}
    \begin{defi}    
        $a:A\to R$ is split by $\sigma:S\to R$ in $\overline{\mathcal{A}}$ when
        \[ \eta^S_{\sigma^*(a)}:\sigma^*(a)\longrightarrow \mathcal{C}_S\mathcal{S}_S(\sigma^*(a)) \]
        is an isomorphism.
    \end{defi}

    \pause \vspace{2mm}
    
    \begin{defi}
        A morphism $\sigma:S\longrightarrow R$ in $\overline{\mathcal{A}}$ is of relative Galois descent with respect to a relatively admissible adjunction when
        \begin{enumerate}
            \item $\sigma^*:\overline{\mathcal{A}}/R\longrightarrow\overline{\mathcal{A}}/S$ is monadic;
            \vspace{1mm}
            \item $\varepsilon^S_f:f\to\mathcal{S}_S\mathcal{C}_S(f)$ is an isomorphism, $\forall f:X\to\mathcal{S}(S)$;
            \vspace{1mm}
            \item $\sigma\circ \mathcal{C}_S(f)\in Split_R(\sigma)$, $\forall f:X\to\mathcal{S}(S)$.
        \end{enumerate}
    \end{defi}
\end{frame}

\begin{frame}{General setting}
    \[\begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&]
        \overline{\mathcal{A}}/R \arrow[rr, "\sigma^*", bend left, ""{name=pull}] \& \& \overline{\mathcal{A}}/S \arrow[ll, "\Sigma_\sigma", bend left, ""{name=comp}] \arrow[rr, "\mathcal{S}_S", bend left, ""{name=Ss}] \& \& \overline{\mathcal{P}}/\mathcal{S}(S) \arrow[ll, "\mathcal{C}_S", bend left, ""{name=Cs}] \\ \\ \\ Split_R(\sigma) \arrow[uuu, tail] \arrow[rr, "\sigma^*", bend left, ""{name=pullres}] \& \& Split_S(1_S) \arrow[uuu, tail] \arrow[rr, bend left, "\mathcal{S}_S", ""{name=Ssequi}] \arrow[ll, "\Sigma_\sigma", bend left, ""{name=compres}] \& \& \overline{\mathcal{P}}/\mathcal{S}(S) \arrow[uuu, Rightarrow, no head] \arrow[ll, "\mathcal{C}_S", bend left, ""{name=Csequi}] \arrow[phantom, from=pull, to=comp, "\dashv" rotate=90] \arrow[phantom, from=pullres, to=compres, "\dashv" rotate=90] \arrow[phantom, from=Ss, to=Cs, "\dashv" rotate=270] \arrow[phantom, from=Ssequi, to=Csequi, "\approx"]
    \end{tikzcd}\]

    %\pause
    
    \begin{center}
        $\mathcal{S}_S\circ \sigma^*:Split_R(\sigma)\longrightarrow\overline{\mathcal{P}}/\mathcal{S}(S)$ is monadic.
    \end{center}
    The functorial part of the monad is the composite:
    \[ T^\sigma=\mathcal{S}_S\circ\sigma^*\circ\Sigma_\sigma\circ \mathcal{C}_S. \]
\end{frame}

\begin{frame}{Kernel pair groupoid}
    Consider the kernel pair $\begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&]
        S\times_RS \arrow[r, "p_1", shift left=1.5] \arrow[r, "p_2"', shift right=1.5] \& S \arrow[r, "\sigma"] \& R
    \end{tikzcd}$ of $\sigma$.

    It induces an internal groupoid $K(\sigma)$ in $\overline{\mathcal{A}}$:
    \begin{center}
        \begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&, center picture]
            (S\times_RS)\times_S(S\times_RS) \arrow[rr, "m"] \& \&  S\times_RS \arrow[rrr, "p_2"', bend right, shift right=2] \arrow[rrr, "p_1", bend left, shift left=2] \arrow["\tau"', loop, distance=2em, in=305, out=235] \& \& \& S \arrow[lll, "\Delta"']
        \end{tikzcd}
    \end{center}
    fixing:
    \begin{enumerate}
        \item $\Delta:=\limitmorph{1_S}{1_S}$;
        \item $\tau:=\limitmorph{p_2}{p_1}$;
        \item $m:=\limitmorph{\pi_1}{\pi_4}$.
    \end{enumerate}
\end{frame}

% 7 %
\begin{frame}{Galois groupoid}
    The functor $\mathcal{S}$ transforms $K(\sigma)$ into an internal groupoid in $\overline{\mathcal{P}}$:
    \small
    \begin{center}
        \begin{tikzcd}[every label/.append style = {font = \footnotesize}, ampersand replacement=\&, center picture]
             \mathcal{S}(S\times_RS)\times_{\mathcal{S}(S)}\mathcal{S}(S\times_RS) \arrow[rr, "\mathcal{S}(m)"] \& \&  \mathcal{S}(S\times_RS) \arrow[rrr, "\mathcal{S}(p_2)"', bend right, shift right=2] \arrow[rrr, "\mathcal{S}(p_1)", bend left, shift left=2] \arrow["\mathcal{S}(\tau)"', loop, distance=2em, in=305, out=235] \& \& \& \mathcal{S}(S) \arrow[lll, "\mathcal{S}(\Delta)"']
        \end{tikzcd}
    \end{center}

    \vspace{3mm}
    
    \begin{defi}
        The image $Gal[\sigma]:=\mathcal{S}(K(\sigma))$ of $K(\sigma)$ through $\mathcal{S}$ is called the Galois groupoid of $\sigma$.
    \end{defi}
\end{frame}

% 8 %
\begin{frame}{Internal covariant presheaves}
    An internal covariant presheaf on $Gal[\sigma]$ in $\overline{\mathcal{P}}$ is a triple $(X,f,\pi)$ where
    
    \vspace{1mm}
    
    \begin{enumerate}
        \item $f:X\longrightarrow \mathcal{S}(S)$
        \item $\pi:\mathcal{S}(S\times_RS)\times_{\mathcal{S}(S)}X\longrightarrow X$
    \end{enumerate}
    with compatibility with the structure of $Gal[\sigma]$.

    \vspace{3mm}
    
    The forgetful functor
    \[
    \begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&, row sep=1pt]
        U:\overline{\mathcal{P}}^{Gal[\sigma]} \arrow[r] \& \overline{\mathcal{P}}/\mathcal{S}(S) \\
        (X,f,\pi) \arrow[r] \& f:X\to \mathcal{S}(S)
    \end{tikzcd}\]
    is monadic. 

    \vspace{1mm}

    The functorial part of the monad is the composite:
    \[ T=\Sigma_{\mathcal{S}(p_2)}\circ{\mathcal{S}(p_1)}^*. \]
\end{frame}

% 9 %
\begin{frame}{Categorical Galois Theorem}
    \begin{teo}
        Let
        $\begin{tikzcd}[ampersand replacement=\&]
            (\mathcal{A},\overline{\mathcal{A}}) \arrow[r, bend left, "\mathcal{S}", ""{name=S}] \& (\mathcal{P},\overline{\mathcal{P}}) \arrow[l, bend left, "\mathcal{C}", ""{name=C}] \arrow[phantom, from=S, to=C, "\dashv" rotate=-90]
        \end{tikzcd}$
        be a relatively admissible adjunction where $\mathcal{A}$ and $\mathcal{P}$ are categories with pullbacks. Let $\sigma:S\to R$ be of relative Galois descent with respect to these data. There exists an equivalence of categories
        \[ Split_R(\sigma)\approx\overline{\mathcal{P}}^{Gal[\sigma]} \]
        between the category of those objects in $\overline{\mathcal{A}}/R$ that are split by $\sigma$ and the category of internal covariant presheaves on the internal groupoid $Gal[\sigma]$ in $\overline{\mathcal{P}}$.
    \end{teo}
\end{frame}

\begin{frame}{Classical Galois correspondence}
    Fix a finite Galois extension $\sigma:K\hookrightarrow L$ of fields.

\end{frame}

\begin{frame}{Grothendieck group of monoids}
    \begin{defi}
        Let $(M,\cdot, 1)$ be a monoid. The Grothendieck group of $M$ is given by
        \[ Gp(M)=\frac{GpF(M)}{N(M)}, \]
        where $GpF(M)$ denotes the free group on $M$ and $N(M)$ the normal subgroup generated by elements of the form $[m_1][m_2][m_1\cdot m_2]^{-1}$.
    \end{defi}

    \pause

    \[\begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&]

    \end{tikzcd}\]
\end{frame}

\begin{frame}{Schreier and homogeneous split epimorphisms}
    Consider a split epimorphism $(f,s)$ of monoid with its kernel:
    \[\begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&]
        K \arrow[rr, tail] \& \& A \arrow[rr, "f", two heads, shift left=1.5] \& \& M. \arrow[ll, "s", shift left=1.5]
    \end{tikzcd}\]
    \begin{defi}
        $(f,s)$ is a Schreier split epimorphism when, for any $a\in A$, there exists a unique $k\in K$ such that $a=k\cdot sf(a)$.
    \end{defi}
    \begin{defi}
        $(f,s)$ is a homogeneous split epimorphism when all the maps
        \[ \mu^r_m:K\longrightarrow f^{-1}(m)\text{, } k \longmapsto k\cdot s(m) \]
        \[ \mu^l_m:K\longrightarrow f^{-1}(m)\text{, }k \longmapsto s(m)\cdot k \]
        are bijective.
    \end{defi}
\end{frame}

% 16 %
\begin{frame}{Special Schreier and special homogeneous surjections}
    Consider a surjection $\sigma:M\rrightarrow N$ of monoid with its kernel pair:
    \[\begin{tikzcd}[every label/.append style = {font = \small}, ampersand replacement=\&]
        M\times_NM \arrow[rr, "p_1", two heads, shift left=3] \arrow[rr, "p_2"', two heads, shift right=3] \& \& M \arrow[ll, "\Delta" description] \arrow[rr, "\sigma", two heads] \& \& N
    \end{tikzcd}\]
    \begin{defi}
        $\sigma:M\rrightarrow N$ is a special Schreier surjection when $(p_1,\Delta)$ is a Schreier split epimorphism.
    \end{defi}
    \begin{defi}
        $\sigma:M\rrightarrow N$ is a special homogeneous surjection when $(p_1,\Delta)$ is a homogeneous split epimorphism.
    \end{defi}
\end{frame}

\begin{frame}{Semidirect form of Schreier split epimorphisms}
    \begin{prop}
